% ---- speech bubble (use inside a tikz [remember picture, overlay] picture) ----
% \bubble{at-point}{pointer-point}{text}
\newcommand{\bubble}[4][text width=4.3cm]{%
  \node[rectangle callout, rounded corners=9pt, fill=wsyellow, draw=none,
        callout absolute pointer={#3}, inner sep=8pt, align=center,
        font=\bfseries\small, #1] at #2 {#4};%
}

% \bubbleat{at-point}{markname}{text} -- bubble whose tail points at a \tikzmark{markname}
% placed inside a listing (use escapeinside, e.g. (*\tikzmark{app}*) ).
\newcommand{\bubbleat}[4][text width=4.3cm]{%
  \node[rectangle callout, rounded corners=9pt, fill=wsyellow, draw=none,
        callout absolute pointer={([xshift=1.1cm]pic cs:#3)},
        inner sep=8pt, align=center, font=\bfseries\small, #1] at #2 {#4};%
}

% \topband{fraction}{title} -- orange band across the top of a [plain] frame,
% with the title (white) at left and the light logo at right. Use inside an overlay.
\newcommand{\topband}[2]{%
  \fill[wsorange] (current page.north west) rectangle ([yshift=-#1\paperheight]current page.north east);
  \node[anchor=north west, text=white, inner sep=0] at \pg{0.85cm}{-0.4cm}
    {\fontsize{27}{33}\selectfont\bfseries #2};
  \node[anchor=north east, inner sep=0] at \pge{-0.85cm}{-0.55cm}
    {\includegraphics[height=0.85cm]{\wsLogoLight}};
}

      \node[inner sep=0] (dia) {\includegraphics[height=0.66\paperheight]{assets/dia-object-model.pdf}};

    \end{tikzpicture}\vfill
