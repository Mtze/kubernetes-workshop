% ---- page-relative coordinate helper for overlays ----

% ---- speech bubble (use inside a tikz [remember picture, overlay] picture) ----
% \bubble{at-point}{pointer-point}{text}
\newcommand{\bubble}[3]{%
  \node[rectangle callout, rounded corners=9pt, fill=wsyellow, draw=none,
        callout absolute pointer={#2},
        inner sep=9pt, align=center, font=\bfseries\small] at #1 {#3};%
}

% ---- full-width orange band (highlight / definition) ----
\newcommand{\orangeband}[1]{%
  \par\smallskip\noindent
  \begin{tikzpicture}
    \node[fill=wsorange, text=white, inner sep=11pt, align=left,
          text width=\dimexpr\textwidth-22pt\relax, font=\large]{#1};%
  \end{tikzpicture}\par\smallskip
}

% ---- title / section / break frames ----
\newcommand{\titleframe}{%

      \fill[wsorange] (current page.north west) rectangle ([yshift=-0.42\paperheight]current page.north east);
      \node[anchor=north west, text=white, inner sep=0, align=left]
        at \pg{0.85cm}{-0.9cm} {\fontsize{32}{38}\selectfont\bfseries \wsTitle};
      \node[anchor=north east, inner sep=0]
        at \pge{-0.85cm}{-0.7cm} {\includegraphics[height=0.95cm]{\wsLogoLight}};

}
